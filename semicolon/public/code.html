import numpy as np
a=np.array([1,402,2,1])   #1차원, 괄호() 안에 들어있는 건 리스트
b=np.array([[2,4,7,8,5],[1,3,7,8,9,],[3,5,7,8,24]])    #2차원
# 여러 차수의 행렬을 만들때, 각각의 방에는 같은 수의 숫자가 있어야한다
c=np.array([[[13,4,5],[1,34,5,]],[[3,5,612],[435,62,76]]])  #3차원
 
list=[3,5,6,8]
d=np.array(list)  #np.array([3,5,6,8])이랑 같은거다.
 
print(a)
print(b[1]) #b의 1번 방 (컴퓨터의 시작은 0부터)
print(c[0][1])  #c의 0번 방 안에 있는 1번째방
print(b[1][0:2]) #b의 1번 방에서 0번에서 2번바로 전까지 가지고 오기

import matplotlib.pyplot as plt
 
years = [1950,1960,1970,1980,1990,2000]
gdp = [67.0,88.0,257.0,1686.0,11684.0,21156.0]
plt.plot(years,gdp, color='green',marker='o', linestyle='solid') #plot에 들어갈 자료와 그래프 색, 그래프 스타일 설정
plt.title('gdp per capita',color='blue')    #제목 설정
plt.xlabel('years')        #x축에 쓰일 자료 설정
plt.ylabel('gdp')        #y축에 쓰일 자료 설정
plt.savefig('파일 이름 적기.png',dpi=600)        #파일 png형식으로 저장하기
plt.show()        #그래프 보여주기
